  ConfigMap         : vault-config
  StatefulSet       : vault-statefulset
  Service           : {{ .Values.vault.service.name }} (headless)
  ClusterRole       : {{ .Release.Name }}-{{ .Values.rbac.clusterRole.name }}
  ClusterRoleBinding: {{ .Release.Name }}-{{ .Values.rbac.clusterRoleBinding.name }}
  ClusterSecretStore: {{ .Release.Name }}-{{ .Values.clusterSecretStore.name }}

Next steps after deployment:
  1. Initialize Vault (first time only):
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault operator init
       (Save the Unseal Keys and Initial Root Token that this command outputs!)

  2. Unseal Vault (Run this 3 times using 3 different unseal keys from step 1):
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault operator unseal <KEY_1>
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault operator unseal <KEY_2>
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault operator unseal <KEY_3>
       (After the 3rd key, the pod status should change to 1/1 Running)

  3. Login and configure the Kubernetes Auth Method for ESO
     Use the Initial Root Token from Step 1 to login:
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault login <ROOT_TOKEN>

     Then enable and configure the kubernetes auth method:
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault auth enable kubernetes

       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- sh -c '
       vault write auth/kubernetes/config \
         kubernetes_host="https://$KUBERNETES_SERVICE_HOST:$KUBERNETES_SERVICE_PORT" \
         kubernetes_ca_cert=@/var/run/secrets/kubernetes.io/serviceaccount/ca.crt \
         issuer="https://kubernetes.default.svc.cluster.local"
       '

       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault write auth/kubernetes/role/{{ .Values.clusterSecretStore.role }} \
           bound_service_account_names={{ .Values.clusterSecretStore.esoServiceAccount.name }} \
           bound_service_account_namespaces={{ .Values.clusterSecretStore.esoServiceAccount.namespace }} \
           policies=eso-policy ttl=24h

  4. Add your application secrets to Vault
     Create the secret engine:
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault secrets enable -version=2 -path=secret kv

     Create the policy for ESO:
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- sh -c 'cat <<EOF > /tmp/eso-policy.hcl
       path "secret/data/one2n/dev/app-config" {
         capabilities = ["read"]
       }
       EOF
       vault policy write eso-policy /tmp/eso-policy.hcl'

     Add the actual secrets (Replace with your actual DB URL, password, and dockerhub config):
       kubectl exec -n {{ .Values.namespace }} vault-statefulset-0 -- vault kv put secret/one2n/dev/app-config \
           db_url="postgresql://postgres:password@postgres-db-svc.student-app.svc.cluster.local:5432/students_db" \
           db_password="password" \
           dockerconfigjson='\''{"auths":{"https://index.docker.io/v1/":{"auth":"base64_encoded_auth"}}}'\''

  5. Vault API:
       http://{{ .Values.vault.service.name }}.{{ .Values.namespace }}.svc.cluster.local:{{ .Values.vault.ports.api }}
